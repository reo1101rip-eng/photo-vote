<script>
const $ = id => document.getElementById(id);

/* ★設定: Apps Script をウェブアプリとしてデプロイしたときの /exec URL を貼ってください */
const API_URL = 'https://script.google.com/macros/s/AKfycbykfY5obhkdRyaFweG95bBACCSBVujRvI5BsKNg4QySM_xH7XXSSJMoOHVK-_sffLXz/exec';

/* Apps Script API を JSONP で呼ぶ（CORS回避のため） */
let cbSeq = 0;
const run = (action, extra) => new Promise((resolve, reject) => {
  if (!API_URL || API_URL.indexOf('script.google.com') === -1) {
    return reject(new Error('API_URL が未設定です。index.html の API_URL を確認してください。'));
  }
  const cb = '__vote_cb_' + Date.now() + '_' + (++cbSeq);
  const script = document.createElement('script');
  const timer = setTimeout(() => { cleanup(); reject(new Error('通信がタイムアウトしました。')); }, 25000);
  function cleanup() { clearTimeout(timer); try { delete window[cb]; } catch (e) { window[cb] = undefined; } if (script.parentNode) script.parentNode.removeChild(script); }
  window[cb] = data => { cleanup(); resolve(data); };
  script.onerror = () => { cleanup(); reject(new Error('通信に失敗しました。')); };
  let src = API_URL
    + '?action=' + encodeURIComponent(action)
    + '&deviceId=' + encodeURIComponent(DID)
    + '&callback=' + cb
    + '&_=' + Date.now();
  if (extra) for (const k in extra) src += '&' + encodeURIComponent(k) + '=' + encodeURIComponent(extra[k]);
  script.src = src;
  document.head.appendChild(script);
});

const KEY = 'orix_photo_vote_device_id';

/* 端末ID: 初回のみ生成してブラウザに保存（1人1回の判定に使う） */
function deviceId() {
  let id = null;
  try { id = localStorage.getItem(KEY); } catch (e) {}
  if (!id) {
    const rnd = (window.crypto && crypto.randomUUID) ? crypto.randomUUID() : (Date.now() + '' + Math.random().toString(36).slice(2) + Math.random().toString(36).slice(2));
    id = rnd.replace(/[^A-Za-z0-9]/g, '').slice(0, 48).padEnd(16, 'x');
    try { localStorage.setItem(KEY, id); } catch (e) {}
  }
  return id;
}
const DID = deviceId();

let photos = [];
let selectedId = null;

function show(id) {
  document.querySelectorAll('main > .view').forEach(v => v.classList.remove('on'));
  $(id).classList.add('on');
  $('voteBar').style.display = (id === 'v-vote') ? 'block' : 'none';
}

function fail(err) {
  $('errMsg').textContent = err && err.message ? err.message : 'エラーが発生しました';
  show('v-err');
}

function showDone(state) {
  if (state && state.already) $('doneTitle').textContent = '投票済みです';
  const p = photos.filter(x => x.id === (state && state.photoId))[0];
  if (p && p.photo) {
    $('donePhoto').src = p.photo;
    $('donePhotoBox').style.display = 'block';
    $('doneLabel').textContent = 'あなたが投票した写真：' + p.title;
  } else if (state && state.title) {
    $('doneLabel').textContent = 'あなたが投票した写真：' + state.title;
  }
  show('v-done');
}

function renderGrid() {
  const grid = $('grid');
  grid.innerHTML = '';
  photos.forEach(p => {
    const d = document.createElement('div');
    d.className = 'card' + (p.id === selectedId ? ' selected' : '');
    d.innerHTML = '<span class="num">' + p.id + '</span><span class="check">✓</span>';
    if (p.photo) {
      const img = document.createElement('img');
      img.src = p.photo; img.alt = p.title; img.loading = 'lazy';
      d.appendChild(img);
    }
    d.addEventListener('click', () => {
      selectedId = p.id;
      document.querySelectorAll('.card').forEach(c => c.classList.remove('selected'));
      d.classList.add('selected');
      $('selLabel').textContent = '選択中：' + p.title;
      $('btnVote').disabled = false;
    });
    grid.appendChild(d);
  });
}

async function init() {
  show('v-loading');
  try {
    const [status, list] = await Promise.all([run('status'), run('photos')]);
    if (status.error) throw new Error(status.error);
    if (list.error) throw new Error(list.error);
    photos = list.photos || [];
    if (!photos.length) throw new Error('写真データがありません。「写真」シートを確認してください。');
    if (status.voted) { showDone(status); return; }
    renderGrid();
    show('v-vote');
  } catch (e) { fail(e); }
}

$('btnVote').addEventListener('click', async () => {
  if (!selectedId) return;
  $('btnVote').disabled = true;
  $('btnVote').textContent = '投票しています…';
  try {
    const res = await run('vote', { photoId: selectedId });
    if (res.error) throw new Error(res.error);
    showDone(res);
  } catch (e) {
    $('btnVote').disabled = false;
    $('btnVote').textContent = 'この写真に投票する';
    fail(e);
  }
});

$('btnRetry').addEventListener('click', init);
init();
